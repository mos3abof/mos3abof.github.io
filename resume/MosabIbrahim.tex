%-------------------------------------------------------------------------------
%	SECTION TEMPLATES — auto-load templates from lib/templates/
%-------------------------------------------------------------------------------
% Raw \directlua (not a luacode environment) so the \input'd definitions stay
% global; string.char avoids the %, backslash and brace catcode traps.
\directlua{
  local dir = "lib/templates"
  for entry in lfs.dir(dir) do
    if entry:sub(-4) == ".tex" then
      tex.sprint(string.char(92) .. "input" .. string.char(123) .. dir .. "/" .. entry .. string.char(125))
    end
  end
}
